<?php

    $students = array(

        array(

            "name" => "alfred",

            "web" => 80,

            "db" => 85,

            "pe" => 70

        ),

        array(

            "name" => "josh",

            "web" => 90,

            "db" => 92,

            "pe" => 97

        ),

        array(

            "name" => "ron",

            "web" => 70,

            "db" => 75,

            "pe" => 77

        ),

    );

    foreach ($students as $student) {

        $avgGrade = 0;

        foreach ($student as $key => $value) {

            if ($key == "name") {

                echo "Name: $key \n";

            } else {

                $avgGrade = $avgGrade + $value;

            }

        }

        echo "Total Average Grade: ". $avgGrade / (count($student) - 1);

        echo "\n\n";

    }

?>
 




<?php

$items = array(
    array(
        "qty" => 2,
        "desc" => "ITEM 1",

    ),
    array(
        "qty" => 7,
        "desc" => "ITEM 2",

    ),
    array(

    ),
    array(
        "qty" => 2,
        "desc" => "ITEM 4",
        "amt" => 20
    )
);

echo "QTY  DESC    AMT   TOTAL\n";

$overallTotal = 0;

foreach ($items as $item) {

    $total = $item["qty"] * $item["amt"];
    
    echo "(".$item["qty"].") ";
    echo $item["desc"]."   ";
    echo $item["amt"]."   ";
    echo $total;
    echo "\n";

    $overallTotal = $overallTotal + $total;
}
